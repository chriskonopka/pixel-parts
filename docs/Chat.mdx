# Chat Helper Functions

Helper functions for the McDermott Chat experience.

## `getConversationHistory`

Builds a **conversation summary string** and a **FileList** from records stored in your client-side IndexedDB database. It expects each record to include a user's question, the assistant's response, and any uploaded files.

### Usage

<Source 
  code={`
    import * as React from 'react';
    import { WebPartContext } from '@microsoft/sp-webpart-base'; 
    import IndexedDBClient, { setDbConfig } from '@mwe-apps/pixel-parts/lib/services';
    import { getConversationHistory } from '@mwe-apps/pixel-parts/lib/helpers';

    export interface ConversationHistory {
        files: FileList;
        conversation: string;
    }

    export interface MyComponent {
      context: WebPartContext
    }

    const MyComponent = (props: MyComponent): React.ReactElement => {
      const [conversationHistory, setConversationHistory] = React.useState<ConversationHistory | undefined>();

      const params = new URLSearchParams(window.location.search);
      const resourceId = params.get('resourceId');

      const getConversationHistoryFromStorage = async (): Promise<ConversationHistory> => {
        try {
            const { dbName, storeName } = setDbConfig(context, resourceId);
            const db = new IndexedDBClient(dbName);
            const history = await getConversationHistory(db, storeName);

            setConversationHistory(history);
        } catch (err) {
            console.error('Could not get conversation history', err);
            // handle error...
        }
      };

      React.useEffect(() => {
        getConversationHistoryFromStorage();
      }, []);

      return (
        <>...</>
      );
    };
  `}

  dark 
  format
/>
